\begin{frame}{SOTA Takeaway}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.50\textwidth}
            \small
            \textbf{Changed-region accuracy}
            \begin{itemize}
                \item DSIFN: ours improves over Mamba-CD by \(+0.06\) F1 and \(+0.02\) IoU.
                \item WHU: MambaRefine-CD improves over Mamba-CD by \(+0.14\) F1 and \(+0.27\) IoU.
                \item WHU VMamba improves over Mamba-CD by \(+0.29\) F1 and \(+0.54\) IoU.
            \end{itemize}
        \end{column}
        \begin{column}{0.47\textwidth}
            \small
            \textbf{Why still useful}
            \begin{itemize}
                \item D-RBI explicitly separates region and boundary evidence.
                \item ARF-FPN adapts receptive field per scale instead of using one fixed context size.
                \item Boundary residual correction is bounded in logit space, so it sharpens edges without rewriting the whole mask.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}
